label: "Inheritance & polymorphism"
example: |-
  interface IShape { double Area(); }

  abstract class Shape : IShape
  {
      public string Name = "";
      public abstract double Area();
  }

  class Circle : Shape
  {
      public double R;
      public Circle(double r) { R = r; Name = "circle"; }
      public override double Area() => 3.14159 * R * R;
  }

  class Square : Shape
  {
      public double S;
      public Square(double s) { S = s; Name = "square"; }
      public override double Area() => S * S;
  }

  Shape[] shapes = { new Circle(2), new Square(3) };
  foreach (var s in shapes)
      System.Console.WriteLine($"{s.Name}: {s.Area():F2}");
