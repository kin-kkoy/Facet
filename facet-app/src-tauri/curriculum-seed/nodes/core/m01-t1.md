label: "Value vs reference"
example: |-
  struct P { public int X; }
  class C { public int X; }

  var a = new P { X = 1 };
  var b = a;        // struct: b is a copy
  b.X = 99;

  var c = new C { X = 1 };
  var d = c;        // class: d is the same object
  d.X = 99;

  System.Console.WriteLine($"struct a.X={a.X} (unchanged), class c.X={c.X} (changed)");
