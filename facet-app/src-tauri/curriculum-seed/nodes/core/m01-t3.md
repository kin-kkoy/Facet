  bool TryHalve(int n, out int result)
  {
      result = n / 2;
      return n % 2 == 0;
  }

  (int min, int max) Range(int[] a) => (a.Min(), a.Max());

  var ok = TryHalve(10, out int h);
  var (lo, hi) = Range(new[] { 5, 2, 9, 1 });
  System.Console.WriteLine($"halve ok={ok} h={h}; range {lo}..{hi}");
