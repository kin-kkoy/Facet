  int[] nums = { 5, 2, 8, 1, 9, 3, 7 };

  var top3odds = nums.Where(n => n % 2 == 1)
                     .OrderByDescending(n => n)
                     .Take(3)
                     .ToList();
  var sum = nums.Where(n => n > 4).Sum();

  System.Console.WriteLine($"top-3 odds: {string.Join(", ", top3odds)}; sum>4: {sum}");
