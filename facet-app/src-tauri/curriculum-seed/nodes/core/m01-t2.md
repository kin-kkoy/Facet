example: |-
  string Describe(int? n) => n switch
  {
      null         => "nothing",
      < 0          => "negative",
      0            => "zero",
      > 0 and < 10 => "small",
      _            => "big"
  };

  foreach (int? x in new int?[] { null, -3, 0, 7, 42 })
      System.Console.WriteLine($"{x}: {Describe(x)}");
