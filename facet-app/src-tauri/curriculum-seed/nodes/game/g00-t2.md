example: |-
  double Length(double x, double y) => System.Math.Sqrt(x * x + y * y);

  (double x, double y) Normalize(double x, double y)
  {
      var l = Length(x, y);
      return l == 0 ? (0, 0) : (x / l, y / l);
  }

  double px = 0, py = 0, vx = 3, vy = 4;
  for (int frame = 0; frame < 5; frame++)
  {
      var (nx, ny) = Normalize(vx, vy);
      px += nx * 2;   // move 2 units/frame along the velocity
      py += ny * 2;
      System.Console.WriteLine($"frame {frame}: pos=({px:F2},{py:F2})");
  }
